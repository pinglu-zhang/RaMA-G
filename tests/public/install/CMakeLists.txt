cmake_minimum_required(VERSION 3.22)
project(RaMAGInstalledPublicConsumer LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(RaMAGPairwise CONFIG REQUIRED)
find_package(RaMAGCentromere CONFIG REQUIRED)
add_executable(public-consumer main.cpp)
target_link_libraries(public-consumer PRIVATE RaMAG::pairwise RaMAG::centromere)
if(EXPECT_BACKEND)
  target_compile_definitions(public-consumer PRIVATE EXPECT_BACKEND=1)
else()
  target_compile_definitions(public-consumer PRIVATE EXPECT_BACKEND=0)
endif()
